# 连接健康评估 - 算法实现逻辑

## 一、模块概述

连接健康评估模块位于`dashboard_widget.py`中，通过综合分析多项性能指标，为TCP连接提供一个**0-100的健康评分**，帮助用户快速评估连接质量。

## 二、评估指标

### 2.1 核心评估指标

| 指标 | 说明 | 影响 |
|------|------|------|
| RTT（往返时延） | 数据包往返时间 | RTT高表示延迟大 |
| 丢包率 | 重传占比 | 丢包高表示网络不稳定 |
| 吞吐量 | 数据传输速率 | 吞吐量低可能有瓶颈 |
| 连接时长 | 连接持续时间 | 过短可能不稳定 |

### 2.2 辅助统计指标

| 指标 | 说明 |
|------|------|
| 总数据包数 | 连接中的数据包总量 |
| 总字节数 | 传输的数据总量 |
| 数据包比例 | 数据包占总包数的比例 |
| 窗口大小 | 平均和最大窗口 |
| 重传次数 | 检测到的重传数量 |

## 三、健康评分算法

### 3.1 评分模型

```python
def calculate_health_score(self, throughput, rtt, loss_rate, duration):
    """
    计算连接健康分数（0-100）
    
    评分模型:
    - 基础分: 100分
    - RTT惩罚: 根据RTT大小扣分
    - 丢包惩罚: 根据丢包率扣分
    - 吞吐量惩罚: 根据吞吐量扣分
    - 连接时长惩罚: 过短扣分
    
    最终分数 = max(0, min(100, 计算结果))
    
    参数:
        throughput: 吞吐量 (KB/s)
        rtt: 平均RTT (ms)
        loss_rate: 丢包率 (%)
        duration: 连接时长 (s)
    
    返回:
        int: 健康分数 (0-100)
    """
    score = 100  # 基础分
    
    # RTT惩罚（RTT越高，扣分越多）
    if rtt > 0:
        if rtt > 200:      # RTT > 200ms
            score -= 30    # 严重延迟
        elif rtt > 100:    # RTT > 100ms
            score -= 20    # 中等延迟
        elif rtt > 50:     # RTT > 50ms
            score -= 10    # 轻微延迟
    
    # 丢包率惩罚（丢包率越高，扣分越多）
    if loss_rate > 10:     # > 10%
        score -= 40        # 严重丢包
    elif loss_rate > 5:    # > 5%
        score -= 25        # 中等丢包
    elif loss_rate > 1:    # > 1%
        score -= 10        # 轻微丢包
    
    # 吞吐量惩罚（相对评估）
    if throughput > 500:   # > 500 KB/s
        score += 0         # 优秀,不扣分
    elif throughput > 100: # > 100 KB/s
        score -= 5         # 良好
    elif throughput > 10:  # > 10 KB/s
        score -= 10        # 一般
    else:                  # <= 10 KB/s
        score -= 15        # 较差
    
    # 连接时长惩罚（过短可能不稳定）
    if duration < 1:       # < 1秒
        score -= 10        # 连接太短,可能不稳定
    
    # 确保分数在0-100范围内
    return max(0, min(100, score))
```

```python
class GaugeWidget(QWidget):
    """仪表盘组件"""
    
    def __init__(self, title="", unit="", max_value=100):
        super().__init__()
        self.title = title
        self.unit = unit
        self.max_value = max_value
        self.current_value = 0
        
    def paintEvent(self, event):
        """绘制仪表盘"""
        painter = QPainter(self)
        painter.setRenderHint(QPainter.Antialiasing)
        
        # 计算中心和半径
        center_x = self.width() / 2
        center_y = self.height() - 40
        radius = min(self.width(), self.height() - 40) / 2 - 10
        
        # 绘制背景圆弧
        rect = QRectF(center_x - radius, center_y - radius, 
                     radius * 2, radius * 2)
        painter.setPen(QPen(QColor(220, 220, 220), 15, Qt.SolidLine, Qt.RoundCap))
        painter.drawArc(rect, 180 * 16, 180 * 16)  # 半圆
        
        # 计算数值对应的角度
        if self.current_value > 0:
            progress = self.current_value / self.max_value
            angle = int(progress * 180 * 16)
            
            # 根据进度选择颜色
            if progress < 0.5:
                color = QColor(76, 175, 80)   # 绿色 - 良好
            elif progress < 0.75:
                color = QColor(255, 193, 7)   # 黄色 - 中等
            else:
                color = QColor(244, 67, 54)   # 红色 - 警告
            
            painter.setPen(QPen(color, 15, Qt.SolidLine, Qt.RoundCap))
            painter.drawArc(rect, 180 * 16, angle)
        
        # 绘制数值
        painter.setPen(QColor(50, 50, 50))
        painter.setFont(QFont("Arial", 24, QFont.Bold))
        painter.drawText(QRectF(0, center_y - 30, self.width(), 30),
                        Qt.AlignCenter, f"{self.current_value:.1f}")
        
        # 绘制单位
        painter.setFont(QFont("Microsoft YaHei", 10))
        painter.drawText(QRectF(0, center_y, self.width(), 20),
                        Qt.AlignCenter, self.unit)
        
        # 绘制标题
        painter.setFont(QFont("Microsoft YaHei", 11, QFont.Bold))
        painter.drawText(QRectF(0, 10, self.width(), 25),
                        Qt.AlignCenter, self.title)
```

```python
def update_view(self, packets, rtt_data=None, retransmissions=None):
    """更新仪表盘视图"""
    if not packets:
        return
    
    # 计算基本统计
    total_packets = len(packets)
    total_bytes = sum(pkt['length'] for pkt in packets)
    data_packets = sum(1 for pkt in packets if pkt['length'] > 0)
    
    start_time = packets[0]['time']
    end_time = packets[-1]['time']
    duration = end_time - start_time
    
    # 窗口统计
    windows = [pkt['window'] for pkt in packets]
    avg_window = sum(windows) / len(windows)
    max_window = max(windows)
    
    # 吞吐量
    throughput_kbps = (total_bytes / duration / 1024) if duration > 0 else 0
    
    # 重传统计
    retrans_count = len(retransmissions) if retransmissions else 0
    
    # RTT统计
    avg_rtt = 0
    if rtt_data:
        rtt_values = [d['rtt'] for d in rtt_data]
        avg_rtt = sum(rtt_values) / len(rtt_values)
    
    # 丢包率估算
    loss_rate = (retrans_count / total_packets * 100) if total_packets > 0 else 0
    
    # 更新仪表盘
    self.throughput_gauge.set_value(throughput_kbps)
    self.rtt_gauge.set_value(avg_rtt)
    self.loss_gauge.set_value(loss_rate)
    
    # 更新卡片
    self.total_packets_card.set_value(total_packets)
    self.total_bytes_card.set_value(f"{total_bytes / 1024:.1f} KB")
    self.duration_card.set_value(f"{duration:.2f} s")
    self.retrans_card.set_value(retrans_count,
                                f"{loss_rate:.2f}% 丢包率")
    
    # 计算健康分数
    health_score = self.calculate_health_score(
        throughput_kbps, avg_rtt, loss_rate, duration
    )
    
    # 更新健康进度条
    self.update_health_bar(health_score)
```

## 六、评分参考标准

### 6.1 RTT参考值

| RTT范围 | 评级 | 说明 |
|---------|------|------|
| < 20ms | 极佳 | 局域网或近距离 |
| 20-50ms | 良好 | 同城或同区域 |
| 50-100ms | 中等 | 跨区域 |
| 100-200ms | 较高 | 跨国 |
| > 200ms | 很高 | 需要优化 |

### 6.2 丢包率参考值

| 丢包率 | 评级 | 说明 |
|--------|------|------|
| < 0.1% | 极佳 | 网络稳定 |
| 0.1-1% | 良好 | 正常范围 |
| 1-5% | 中等 | 可接受 |
| 5-10% | 较高 | 影响体验 |
| > 10% | 很高 | 需要修复 |

### 6.3 吞吐量参考

吞吐量评估需要考虑应用场景：
- **网页浏览**: > 100 KB/s 良好
- **视频流**: > 500 KB/s 良好
- **文件下载**: 取决于带宽
