	}

}

void delete_DLnode(DLinklist& DL,int pos) {
	int loc = 0;
	DLnode* p = DL;
	while (p->next != NULL) {
		p = p->next;
		loc++;
		if (loc == pos) {
			break;
		}
	}
	if (p->next == NULL) {
		free(p);
	}
	else if (p->next->next == NULL) {
		DLnode* s = p->next;
		p->data = s->data;
		p->next = NULL;
		free(s);
	}
	DLnode* s = p->next;
	p->data = s->data;
	p->next = s->next;
	s->next->prior = p;
	delete s;
}

typedef struct Tnode {

void Creat_BiTree(BiTree& T) {

		T = nullptr;

void Pre(BiTree T) {
	if (T) {
		visit(T);
		Pre(T->lchild);
		Pre(T->rchild);
	}

void In(BiTree T) {
	if (T) {
		In(T->lchild);
		visit(T);
		In(T->rchild);

		visit(T);

int get_high(BiTree T) {
	if (T == NULL) {
		return 0;
	}
	int lhigh = get_high(T->lchild);
	int rhigh = get_high(T->rchild);
	int high = (lhigh > rhigh ? lhigh : rhigh) + 1;
	return high;
}

void Swap(BiTree& T1, BiTree& T2)
{
	Tnode* t = T1;
	T1 = T2;
	T2 = t;
}

void NodeSwap(BiTree& T)
{
	if (T != NULL)
	{
		if (T->lchild != NULL && T->rchild != NULL)
		{
			Swap(T->lchild, T->rchild);
		}
		else if (T->lchild != NULL && T->lchild == NULL)
		{
			T->rchild = T->lchild;
			T->lchild = NULL;
		}
		else if (T->lchild == NULL && T->rchild != NULL) {
			T->lchild = T->rchild;
			T->rchild = NULL;
		}
		NodeSwap(T->lchild);
		NodeSwap(T->rchild);
	}
}

bool same(BiTree T1, BiTree T2) {
	if (!T1 && !T2) {
		return true; 
	}
	if (!T1 || !T2) {
		return false;
	}
	if (T1->data != T2->data) {
		return false; 
	}
	return same(T1->lchild, T2->lchild) && same(T1->rchild, T2->rchild);
}

Tnode* LCA(Tnode* root, Tnode* p, Tnode* q) {
	if (root == q || root == p || root == NULL) return root;
	Tnode* left = LCA(root->lchild, p, q);

	if (left != NULL && right != NULL) return root;
	if (left == NULL && right != NULL) return right;
	else if (left != NULL && right == NULL) return left;
	else return NULL;
}

#include <iostream>

struct MGraph {
    int vex[100] = { 0 };
    int edge[100][100] = { {0} };

    int MG_nextadj(int vex, int w);

struct Arcnode {
    int vex;
    struct Arcnode* nextarc;
};

struct Vexnode {
    int vex;
    Arcnode* firstarc;
};

struct LGraph {
    Vexnode Vexlist[100];

void Creat_MGraph(MGraph& mg) {
    cout << "输入顶点的个数:";

    while (i != -1 && j != -1) {
        cout << "输入边，以-1 -1结束：";

}

void Creat_LGraph(LGraph& lg) {
    cout << "输入顶点的个数:";
    cin >> lg.numvex;
    cout << "输入弧的个数：";
    cin >> lg.numarc;
    for (int i = 0; i < lg.numvex; i++) {
        lg.Vexlist[i].vex = i;
        lg.Vexlist[i].firstarc = nullptr;
    }
    int i = 0, j = 0;
    while (i != -1 && j != -1) {
        cout << "输入边，以-1 -1结束:";
        cin >> i >> j;
        Arcnode* s = new Arcnode;
        s->vex = j;
        s->nextarc = lg.Vexlist[i].firstarc;
        lg.Vexlist[i].firstarc = s;
        s = new Arcnode;
        s->vex = i;
        s->nextarc = lg.Vexlist[j].firstarc;
        lg.Vexlist[j].firstarc = s;
    }
}

int MGraph::MG_firstadj(int vex) {
    if (vex != -1) {
        for (int i = 0; i < numvex; i++) {
            if (edge[vex][i] == 1) {
                return i;
            }
        }
    }
    return -1;
}

int MGraph::MG_nextadj(int vex, int w) {
    if (vex != -1 && w != -1) {
        for (int i = w + 1; i < numvex; i++) {
            if (edge[vex][i] == 1) {
                return i;
            }
        }
    }
    return -1;
}

int LG_firstadj(LGraph lg, int vex) {
    if (vex != -1) {
        Arcnode* s = lg.Vexlist[vex].firstarc;
        if (s != NULL) {
            return s->vex;
        }
    }
    return -1;
}

int LG_nextadj(LGraph lg, int vex, int w) {
    if (vex != -1 && w != -1) {
        Arcnode* s = lg.Vexlist[vex].firstarc;
        while (s != NULL && s->vex != w) {
            s = s->nextarc;
        }
        if (s != NULL && s->nextarc != NULL) {
            return s->nextarc->vex;
        }
    }
    return -1;
}

void MG_DFS(MGraph& mg, int v) {
    cout << v << " ";
    mg.MG_visited[v] = true;
    for (int w = mg.MG_firstadj(v); w >= 0; w = mg.MG_nextadj(v, w)) {
        if (!mg.MG_visited[w]) {
            MG_DFS(mg, w);
        }
    }
}

int dfn[100] = { 0 };
int low[100] = { 0 };
int num = 0;

void tarjan(MGraph mg, int v, int parent) {
    dfn[v] = low[v] = ++num;
    int childCount = 0; 
    bool isgedian = false; 
    for (int w = mg.MG_firstadj(v); w >= 0; w = mg.MG_nextadj(v, w)) {
        if (!dfn[w]) {
            childCount++;
            tarjan(mg, w, v);
            low[v] = min(low[v], low[w]);
            if (low[w] >= dfn[v]) {
                isgedian = true;
            }
        }
    }
    if ((parent == -1 && childCount > 1) || (parent != -1 && isgedian)) {
        cout << "Node " << v << "是割点";
    }
}

void Destroy_LGraph(LGraph& lg) {
    for (int i = 0; i < lg.numvex; i++) {
        Arcnode* cur = lg.Vexlist[i].firstarc;
        while (cur != nullptr) {
            Arcnode* next = cur->nextarc;
            delete cur; 
            cur = next;
        }
    }
}

int getarc(MGraph mg, int vex) {
    int sum = 0;
    for (int j = 0; j < mg.numvex; j++) {
        if (mg.edge[vex][j]) {
            sum++;
        }
    }
    return sum;
}

int getmax(MGraph mg) {
    int maxDegree = 0; 
    int maxVertex = -1; 
    for (int i = 0; i < mg.numvex; i++) {
        int degree = getarc(mg, i); 
        if (degree > maxDegree) {
            maxDegree = degree;
            maxVertex = i;
        }
    }
    return maxVertex; 
}
