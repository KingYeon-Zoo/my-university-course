    return true;
}

#include<iostream>

typedef struct Dnode {
    int data;
    Dnode* prior, * next;
} *DLinkList;

void Creat_Empty_List(DLinkList& L) {
    L = new Dnode;
    L->prior = NULL;
    L->next = NULL;
}

DLinkList DList_TailInsert(DLinkList& L) {
    int x = 0;
    Creat_Empty_List(L);
    Dnode* r = L;
    while (x != -1) {
        cout << "Enter x, -1 stop:";
        cin >> x;
        if (x != -1) {
            Dnode* s = new Dnode;
            s->data = x;
            s->next = NULL;
            s->prior = r;
            r->next = s;
            r = s;
        }
    }
    r->next = L;
    L->prior = r;
    return L;
}

void Print(DLinkList L) {
    Dnode* p = L->next;
    while (p != L) {
        cout << p->data << " ";
        p = p->next;
    }
}

    int sum = 0;
    Dnode* p = L->next;
    while (p != L) {
        sum++;
        p = p->next;
    }
    return sum;
}

bool duichen(DLinkList L, int x) {
    if (L == nullptr || L->next == L) {
        return true;
    }
    Dnode* p = L->next;
    Dnode* q = L->prior;
    if (x % 2 == 1) {        
        while (p != q) {
            if (p->data != q->data) {
                return false;
            }
            p = p->next;
            q = q->prior;
        }
    }
    else {       
        while (p->next != q) {
            if (p->data != q->data) {
                return false;
            }
            p = p->next;
            q = q->prior;
        }
    }
    return true;
}
