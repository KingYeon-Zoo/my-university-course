# 指令调度优化对比

## 一、核心优化技术说明

### 1. RAW (Read After Write) 相关消除

**问题**：后一条指令读取前一条指令写入的寄存器，需要等待前一条指令完成。

**解决**：在两条相关指令之间插入独立的指令。

### 2. Load-Use 相关消除

**问题**：Load指令后立即使用加载的数据，需要等待内存访问完成（通常2-3个周期）。

**解决**：在load指令和使用指令之间插入其他独立指令。

### 3. 分支延迟槽利用

**问题**：MIPS的分支指令后有一个延迟槽，如果空着会浪费一个周期。

**解决**：将有用的指令移到延迟槽中执行。

## 二、代码对比

### 2.1 print_loop 循环优化

#### 原始代码（存在多个停顿）：

```assembly
print_loop:
        slt r6,r5,r17       ; 周期1: 比较
        beqz r6,start_sum   ; 周期2: 分支 [需要等待r6，停顿1周期]
                            ; 周期3: 延迟槽空 [浪费1周期]
        
        dsll r7,r5,3        ; 周期4: 计算偏移
        dadd r8,r16,r7      ; 周期5: 地址加法 [需要等待r7，停顿1周期]
        ld r9,(r8)          ; 周期6: 加载 [需要等待r8，停顿1周期]
        
        daddi r24,r0,1      ; 周期7: 准备输出码
        sd r9,(r22)         ; 周期8-10: 存储 [需要等待r9，停顿2-3周期]
        sd r24,(r21)        ; 周期11: 控制
        
        daddi r5,r5,1       ; 周期12: i++
        j print_loop        ; 周期13: 跳转
                            ; 周期14: 延迟槽空 [浪费1周期]

; 总周期：约14周期，包含6-7个周期的停顿
```

#### 优化代码（停顿大幅减少）：

```assembly
print_loop:
        slt r6,r5,r17       ; 周期1: 比较
        dsll r7,r5,3        ; 周期2: 计算偏移 [独立指令，消除停顿]
        beqz r6,start_sum   ; 周期3: 分支 [r6已就绪，无停顿]
        dadd r8,r16,r7      ; 周期4: 地址加法 [延迟槽，r7已就绪]
        
        ld r9,(r8)          ; 周期5: 加载 [r8已就绪，无停顿]
        daddi r5,r5,1       ; 周期6: i++ [独立指令，隐藏load延迟]
        
        sd r9,(r22)         ; 周期7: 存储 [r9已就绪，无停顿]
        sd r24,(r21)        ; 周期8: 控制 [r24在循环外准备]
        
        j print_loop        ; 周期9: 跳转
        nop                 ; 周期10: 延迟槽

; 总周期：约10周期，几乎无停顿
; 性能提升：约40%（14周期 → 10周期）
```

### 2.2 array_sum 循环优化

#### 原始代码：

```assembly
sum_loop:
        slt r3,r2,r17       ; 周期1: i < N ?
        beqz r3,end_sum     ; 周期2: 判断 [等待r3，停顿1周期]
                            ; 周期3: 延迟槽空
        
        dsll r4,r2,3        ; 周期4: i * 8
        dadd r5,r16,r4      ; 周期5: 地址计算 [等待r4，停顿1周期]
        ld r6,(r5)          ; 周期6: 加载 [等待r5，停顿1周期]
        
        dadd r18,r18,r6     ; 周期7-9: 求和 [等待r6，停顿2-3周期]
        
        daddi r2,r2,1       ; 周期10: i++
        j sum_loop          ; 周期11: 跳转
                            ; 周期12: 延迟槽空

; 每次迭代：约12周期（包含5-6周期停顿）
; 12次迭代：约144周期
```

#### 简单优化版本：

```assembly
sum_loop:
        slt r3,r2,r17       ; 周期1: i < N ?
        dsll r4,r2,3        ; 周期2: i * 8 [消除停顿]
        beqz r3,end_sum     ; 周期3: 判断 [r3已就绪]
        dadd r5,r16,r4      ; 周期4: 地址计算 [延迟槽，r4已就绪]
        
        ld r6,(r5)          ; 周期5: 加载 [r5已就绪]
        daddi r2,r2,1       ; 周期6: i++ [隐藏load延迟]
        
        dadd r18,r18,r6     ; 周期7: 求和 [r6已就绪]
        
        j sum_loop          ; 周期8: 跳转
        nop                 ; 周期9: 延迟槽

; 每次迭代：约9周期（几乎无停顿）
; 12次迭代：约108周期
; 性能提升：约33%（144 → 108周期）
```

#### 4路展开优化版本：

```assembly
unrolled_loop:
        slt r3,r2,r11       ; 周期1: i < N-余数 ?
        dsll r4,r2,3        ; 周期2: i * 8
        beqz r3,remainder   ; 周期3: 判断
        dadd r5,r16,r4      ; 周期4: 基地址+偏移 [延迟槽]
        
        ; 第1次：加载 A[i]
        ld r6,(r5)          ; 周期5
        
        ; 第2次：计算并加载 A[i+1]
        daddi r7,r4,8       ; 周期6
        dadd r8,r16,r7      ; 周期7
        ld r9,(r8)          ; 周期8
        
        ; 第3次：计算并加载 A[i+2]
        daddi r7,r4,16      ; 周期9
        dadd r8,r16,r7      ; 周期10
        dadd r18,r18,r6     ; 周期11 [r6已就绪]
        ld r6,(r8)          ; 周期12
        
        ; 第4次：计算并加载 A[i+3]
        daddi r7,r4,24      ; 周期13
        dadd r8,r16,r7      ; 周期14
        dadd r18,r18,r9     ; 周期15 [r9已就绪]
        ld r9,(r8)          ; 周期16
        
        ; 完成4次求和
        dadd r18,r18,r6     ; 周期17
        dadd r18,r18,r9     ; 周期18
        
        daddi r2,r2,4       ; 周期19: i += 4
        j unrolled_loop     ; 周期20
        nop                 ; 周期21

; 每4次迭代：约21周期
; 12次迭代（3轮）：约63周期
; 性能提升：约2.3倍（144 → 63周期）
```

## 四、关键优化点详解

### 4.1 寄存器相关消除示例

**原始：**
```assembly
slt r3,r2,r17       ; 产生r3
beqz r3,end_sum     ; 立即使用r3 → 停顿1周期
```

**优化：**
```assembly
slt r3,r2,r17       ; 产生r3
dsll r4,r2,3        ; 独立指令，填充停顿
beqz r3,end_sum     ; 使用r3，无停顿
```

### 4.2 Load-Use延迟隐藏

**原始：**
```assembly
ld r6,(r5)          ; Load需要2-3周期
dadd r18,r18,r6     ; 立即使用 → 停顿2-3周期
```

**优化：**
```assembly
ld r6,(r5)          ; Load需要2-3周期
daddi r2,r2,1       ; 填充1个周期
nop                 ; 如需要，再填充
dadd r18,r18,r6     ; 使用r6，停顿减少或消除
```

### 4.3 分支延迟槽利用

**原始：**
```assembly
beqz r3,end_sum     ; 分支
                    ; 延迟槽空 → 浪费1周期
dsll r4,r2,3        ; 下一条指令
```

**优化：**
```assembly
beqz r3,end_sum     ; 分支
dadd r5,r16,r4      ; 延迟槽中执行有用指令
```

## 六、学习要点

1. **数据相关识别**：找出代码中的RAW、WAR、WAW相关
2. **独立指令识别**：找出可以移动的独立指令
3. **代码重排序**：在不改变语义的前提下重新排列指令
4. **延迟槽填充**：充分利用MIPS的分支延迟槽
5. **循环优化**：展开、软件流水等高级技术

这些技术是现代编译器优化的基础！
